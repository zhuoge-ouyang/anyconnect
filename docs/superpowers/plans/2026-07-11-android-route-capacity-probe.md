Use a BigInteger representation so IPv4 and IPv6 share one audited path:

~~~kotlin
enum class IpFamily(val bitCount: Int, val byteCount: Int) {
    IPV4(32, 4),
    IPV6(128, 16)
}

data class Cidr(
    val network: BigInteger,
    val prefixLength: Int,
    val family: IpFamily
) : Comparable<Cidr> {
    init {
        require(prefixLength in 0..family.bitCount)
        require(network.signum() >= 0)
        require(network <= family.maxValue())
        require(network == network.and(mask(family, prefixLength)))
    }

    val size: BigInteger get() = BigInteger.ONE.shiftLeft(family.bitCount - prefixLength)
    val end: BigInteger get() = network + size - BigInteger.ONE

    fun contains(value: BigInteger): Boolean = value in network..end
    fun overlaps(other: Cidr): Boolean =
        family == other.family && network <= other.end && other.network <= end

    fun address(offset: BigInteger = BigInteger.ZERO): InetAddress {
        require(offset.signum() >= 0 && offset < size)
        return InetAddress.getByAddress((network + offset).toFixedBytes(family.byteCount))
    }

    override fun compareTo(other: Cidr): Int =
        compareValuesBy(this, other, Cidr::family, Cidr::network, Cidr::prefixLength)

    override fun toString(): String = address().hostAddress + "/" + prefixLength

    companion object {
        fun parse(raw: String): Cidr {
            val parts = raw.trim().split("/")
            require(parts.size == 2) { "CIDR must contain one slash: $raw" }
            val address = InetAddress.getByName(parts[0])
            val family = if (address.address.size == 4) IpFamily.IPV4 else IpFamily.IPV6
            val prefix = parts[1].toInt()
            require(prefix in 0..family.bitCount)
            val value = BigInteger(1, address.address)
            return Cidr(value.and(mask(family, prefix)), prefix, family)
        }
    }
}

fun IpFamily.maxValue(): BigInteger = BigInteger.ONE.shiftLeft(bitCount) - BigInteger.ONE

private fun mask(family: IpFamily, prefix: Int): BigInteger {
    if (prefix == 0) return BigInteger.ZERO
    val hostBits = family.bitCount - prefix
    return family.maxValue().xor(BigInteger.ONE.shiftLeft(hostBits) - BigInteger.ONE)
}

private fun BigInteger.toFixedBytes(length: Int): ByteArray {
    val raw = toByteArray()
    val unsigned = if (raw.size > length) raw.copyOfRange(raw.size - length, raw.size) else raw
    return ByteArray(length - unsigned.size) + unsigned
}
~~~

- [ ] **Step 4: Implement collapse and complement**

Use merged inclusive ranges, then convert every gap to minimal CIDRs:

~~~kotlin
object CidrMath {
    private data class Range(val start: BigInteger, val end: BigInteger)

    fun collapse(cidrs: List<Cidr>): List<Cidr> {
        if (cidrs.isEmpty()) return emptyList()
        val family = cidrs.first().family
        require(cidrs.all { it.family == family })
        val merged = mutableListOf<Range>()
        for (cidr in cidrs.sorted()) {
            val previous = merged.lastOrNull()
            if (previous == null || cidr.network > previous.end + BigInteger.ONE) {
                merged += Range(cidr.network, cidr.end)
            } else if (cidr.end > previous.end) {
                merged[merged.lastIndex] = previous.copy(end = cidr.end)
            }
        }
        return merged.flatMap { rangeToCidrs(it.start, it.end, family) }
    }

    fun complement(excluded: List<Cidr>, family: IpFamily): List<Cidr> {
        val normalized = collapse(excluded.filter { it.family == family })
        val result = mutableListOf<Cidr>()
        var cursor = BigInteger.ZERO
        for (cidr in normalized) {
            if (cursor < cidr.network) {
                result += rangeToCidrs(cursor, cidr.network - BigInteger.ONE, family)
            }
            if (cidr.end == family.maxValue()) return result
            cursor = cidr.end + BigInteger.ONE
        }
        if (cursor <= family.maxValue()) {
            result += rangeToCidrs(cursor, family.maxValue(), family)
        }
        return result
    }

    private fun rangeToCidrs(
        start: BigInteger,
        end: BigInteger,
        family: IpFamily
    ): List<Cidr> {
        require(start <= end)
        val result = mutableListOf<Cidr>()
        var cursor = start
        while (cursor <= end) {
            val alignmentBits = if (cursor == BigInteger.ZERO) {
                family.bitCount
            } else {
                cursor.lowestSetBit.coerceAtMost(family.bitCount)
            }
            val remainingBits = (end - cursor + BigInteger.ONE).bitLength() - 1
            val hostBits = minOf(alignmentBits, remainingBits)
            result += Cidr(cursor, family.bitCount - hostBits, family)
            cursor += BigInteger.ONE.shiftLeft(hostBits)
        }
        return result
    }
}
~~~

- [ ] **Step 3: Implement TunCapture**

TunCapture runs one blocking reader thread, stores destination plus the packet bytes needed to match a per-probe nonce, and treats fd closure as normal termination. Its constructor takes ownership of the ParcelFileDescriptor; ProbeRunner must not close the same descriptor separately.

~~~kotlin
class TunCapture(private val pfd: ParcelFileDescriptor) : Closeable {
    private val packets = ConcurrentLinkedQueue<CapturedPacket>()
    private val running = AtomicBoolean(true)
    private val executor = Executors.newSingleThreadExecutor()
    private val future = executor.submit {
        val input = FileInputStream(pfd.fileDescriptor)
        val buffer = ByteArray(65_535)
        while (running.get()) {
            try {
                val count = input.read(buffer)
                if (count < 0) break
                PacketDestinationParser.destination(buffer, count)?.let {
                    packets += CapturedPacket(it.hostAddress, buffer.copyOf(count))
                }
            } catch (_: IOException) {
                if (running.get()) throw
                break
            }
        }
    }

    fun await(destination: InetAddress, nonce: ByteArray, timeoutMs: Long): Boolean {
        val deadline = SystemClock.elapsedRealtime() + timeoutMs
        while (SystemClock.elapsedRealtime() < deadline) {
            if (captured(destination, nonce)) return true
            Thread.sleep(25)
        }
        return captured(destination, nonce)
    }

    fun captured(destination: InetAddress, nonce: ByteArray): Boolean =
        packets.any {
            it.destination == destination.hostAddress &&
                it.bytes.indexOfSubsequence(nonce) >= 0
        }

    override fun close() {
        running.set(false)
        pfd.close()
        future.cancel(true)
        executor.shutdownNow()
    }
}

data class CapturedPacket(val destination: String, val bytes: ByteArray)

private fun ByteArray.indexOfSubsequence(needle: ByteArray): Int {
    if (needle.isEmpty()) return 0
    for (start in 0..size - needle.size) {
        var matches = true
        for (offset in needle.indices) {
            if (this[start + offset] != needle[offset]) {
                matches = false
                break
            }
        }
        if (matches) return start
    }
    return -1
}
~~~

~~~kotlin
enum class DirectResponseStatus { PASS, INCONCLUSIVE }

data class DirectProbeResult(
    val status: DirectResponseStatus,
    val wireNonce: ByteArray,
    val detail: String
)

object UdpPathProbe {
    fun verifyDirectDns(address: InetAddress, nonce: ByteArray): DirectProbeResult {
        require(nonce.size == 16)
        val transactionId =
            ((nonce[0].toInt() and 0xff) shl 8) or (nonce[1].toInt() and 0xff)
        val label = "probe-" + nonce.joinToString("") {
            "%02x".format(it.toInt() and 0xff)
        }
        val wireNonce = label.toByteArray(Charsets.US_ASCII)
        val queryType = if (address.address.size == 4) 1 else 28
        val query = buildDnsQuery(transactionId, "$label.invalid", queryType)
        return try {
            DatagramSocket().use { socket ->
                socket.soTimeout = 1_500
                socket.connect(InetSocketAddress(address, 53))
                socket.send(DatagramPacket(query, query.size))
                val response = ByteArray(2_048)
                val packet = DatagramPacket(response, response.size)
                socket.receive(packet)
                if (responseMatches(response, packet.length, transactionId)) {
                    DirectProbeResult(
                        DirectResponseStatus.PASS,
                        wireNonce,
                        "matching DNS response"
                    )
                } else {
                    DirectProbeResult(
                        DirectResponseStatus.INCONCLUSIVE,
                        wireNonce,
                        "DNS transaction ID mismatch"
                    )
                }
            }
        } catch (error: IOException) {
            DirectProbeResult(
                DirectResponseStatus.INCONCLUSIVE,
                wireNonce,
                error.javaClass.simpleName + ": " + error.message
            )
        }
    }

    internal fun responseMatches(response: ByteArray, length: Int, id: Int): Boolean {
        if (length < 2) return false
        val responseId =
            ((response[0].toInt() and 0xff) shl 8) or
                (response[1].toInt() and 0xff)
        return responseId == id
    }

    internal fun buildDnsQuery(id: Int, name: String, type: Int): ByteArray {
        val output = ByteArrayOutputStream()
        DataOutputStream(output).use { data ->
            data.writeShort(id)
            data.writeShort(0x0100)
            data.writeShort(1)
            data.writeShort(0)
            data.writeShort(0)
            data.writeShort(0)
            name.split('.').forEach { label ->
                val bytes = label.toByteArray(Charsets.US_ASCII)
                require(bytes.size in 1..63)
                data.writeByte(bytes.size)
                data.write(bytes)
            }
            data.writeByte(0)
            data.writeShort(type)
            data.writeShort(1)
        }
        return output.toByteArray()
    }
}
~~~
